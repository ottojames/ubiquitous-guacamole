    2. Create placeholder-to-field mapping:
       ```typescript
       function getPlaceholderValue(token: string, data: NoticeFormData): string {
         const mappings: Record<string, () => string> = {
           'APPLICANT_NAME': () => data.applicant.name,
           'APPLICANT_ADDRESS': () => formatAddress(data.applicant.address),
           'PREMISES_NAME': () => data.premises.name,
           'PREMISES_ADDRESS': () => formatAddress(data.premises.address),
           'AUTHORITY_NAME': () => data.consultation.authority,
           'DEADLINE_DATE': () => formatDate(data.representation_deadline),
           'LICENSABLE_ACTIVITIES': () => data.licensing.activities.join(', '),
           // Add all placeholder tokens from placeholders.ts
         };
         return mappings[token]?.() || `{{${token}}}`; // Keep unknown tokens visible
       }
       ```

    3. Main render function:
       ```typescript
       export function renderTemplateText(templateText: string, data: NoticeFormData): string {
         if (!templateText) return '';

         return templateText.replace(/\{\{(\w+)\}\}/g, (match, token) => {
           const value = getPlaceholderValue(token, data);
           return value || match; // Keep placeholder if no value
         });
       }
       ```

    4. Helper functions:
       - formatAddress(): Combine address fields into single string
       - formatDate(): Convert ISO date to readable format (1 November 2025)
