#### 14. No Timeout Handling for AI Summary API
**Location:** `NewPublishFlow.tsx:391-412`
**Severity:** High
**Category:** Functionality
**Perspective:** UI/UX Developer
**Issue:** The AI summary API call has no timeout. If the API is slow or hangs, users are left waiting indefinitely with no feedback.

**User Impact:** High - poor user experience, perceived application freeze.

**Recommendation:**
```typescript
const controller = new AbortController();
const timeoutId = setTimeout(() => controller.abort(), 10000); // 10s timeout

try {
  const response = await fetch('/api/ai-summary', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ text }),
    signal: controller.signal,
  });
  clearTimeout(timeoutId);

  if (response.ok) {
    const data = await response.json();
    aiSummary = data.summary || "";
  } else {
    console.warn('[AI_SUMMARY] API failed:', response.status);
    aiSummary = generateNoticeSummary(text);
    toast.warning('Using basic summary - AI service unavailable');
  }
} catch (error) {
  clearTimeout(timeoutId);
  if (error.name === 'AbortError') {
    console.warn('[AI_SUMMARY] Request timed out');
    toast.warning('Summary generation timed out - using fallback');
  }
  aiSummary = generateNoticeSummary(text);
}
```

const extractPostcode = (address: string): string | null => {
  const postcodeRegex = /\b[A-Z]{1,2}\d{1,2}[A-Z]?\s?\d[A-Z]{2}\b/i;

const validateAddressHasPostcode = (address: string): boolean => {
